{{- /* Numbered pagination with a sliding window. Expects a page that has called .Paginate, and renders nothing for a single page of results. */ -}} {{- $p := .Paginator -}} {{- if gt $p.TotalPages 1 -}} {{- $current := $p.PageNumber -}} {{- $total := $p.TotalPages -}} {{- $window := 1 -}}
  • {{ if $p.HasPrev }} ← {{ else }} ← {{ end }}
    • {{ range $p.Pagers }} {{ $n := .PageNumber }} {{ $edge := or (eq $n 1) (eq $n $total) }} {{ $diff := sub $n $current }} {{ $near := and (ge $diff (sub 0 $window)) (le $diff $window) }} {{ if or $edge $near }}
  • {{ $n }}
    • {{ else if or (eq $n 2) (eq $n (sub $total 1)) }}
  • …
    • {{ end }} {{ end }}
  • {{ if $p.HasNext }} → {{ else }} → {{ end }}
{{- end -}}
